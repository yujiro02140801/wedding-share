// ------------------------- // 複数アップロード対応（ローディング付き） // ------------------------- window.upload = async () => { const files = document.getElementById("fileInput").files; const uploadBtn = document.querySelector(".upload-box button"); if (!files.length) return alert("ファイルを選んでください"); // ボタンをローディング状態にする uploadBtn.disabled = true; uploadBtn.style.opacity = "0.6"; uploadBtn.innerText = "アップロード中…"; let count = 1; const total = files.length; for (const file of files) { uploadBtn.innerText = `アップロード中… (${count}/${total})`; const filePath = `uploads/${Date.now()}_${file.name}`; const { error } = await supabase.storage .from("wedding-share") .upload(filePath, file); if (error) { alert("アップロード失敗: " + error.message); uploadBtn.disabled = false; uploadBtn.style.opacity = "1"; uploadBtn.innerText = "アップロード"; return; } count++; } alert("アップロード完了！"); // ボタンを元に戻す uploadBtn.disabled = false; uploadBtn.style.opacity = "1"; uploadBtn.innerText = "アップロード"; };
